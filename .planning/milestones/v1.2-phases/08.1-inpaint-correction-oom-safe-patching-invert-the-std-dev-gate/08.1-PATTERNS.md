    min_context: float = 0.25,        # D-06: grown to ≥25% of page dims per axis where room
    margin: float = 0.10,             # D-06: ~10% margin around the mask where room

    (D-07 — no margin where region touches page edge, never defer for edge proximity),
    defer interior edge-straddlers (D-08) — collect deferred, loop until covered.

    # Per region: grow bbox to max(bbox_dim, 0.25*page_dim) per axis where room,

    #   the patch boundary in the INTERIOR is excluded from this patch (deferred list).
